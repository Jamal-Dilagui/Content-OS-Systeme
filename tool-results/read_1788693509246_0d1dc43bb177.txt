   253→// ---------- Main view ----------
   254→export function PinterestView() {
   255→  const { toast } = useToast();
   256→  const qc = useQueryClient();
   257→
   258→  const { data: stats, isLoading: statsLoading } = useQuery<Stats>({
   259→    queryKey: ["pinterest", "stats"],
   260→    queryFn: async () => {
   261→      const res = await fetch("/api/pinterest/stats");
   262→      if (!res.ok) throw new Error("Failed to load stats");
   263→      return res.json();
   264→    },
   265→  });
   266→
   267→  const { data: accounts, isLoading: accountsLoading } = useQuery<Account[]>({
   268→    queryKey: ["pinterest", "accounts"],
   269→    queryFn: async () => {
   270→      const res = await fetch("/api/pinterest/accounts");
   271→      if (!res.ok) throw new Error("Failed to load accounts");
   272→      return res.json();
   273→    },
   274→  });
   275→
   276→  // Advance rotation mutation
   277→  const advance = useMutation({
   278→    mutationFn: async (id: string) => {
   279→      const res = await fetch(`/api/pinterest/accounts/${id}/advance`, {
   280→        method: "POST",
   281→      });
   282→      if (!res.ok) throw new Error("Failed to advance");
   283→      return res.json();
   284→    },
   285→    onSuccess: (data) => {
   286→      qc.invalidateQueries({ queryKey: ["pinterest"] });
   287→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   288→      toast({
   289→        title: "Rotation advanced",
   290→        description: data.newAccountOfDay
   291→          ? `Now working: ${data.newAccountOfDay.name}`
   292→          : "No next active account — add or resume one.",
   293→      });
   294→    },
   295→    onError: () => toast({ title: "Failed to advance rotation", variant: "destructive" }),
   296→  });
   297→
   298→  // Update account mutation
   299→  const updateAccount = useMutation({
   300→    mutationFn: async ({
   301→      id,
   302→      data,
   303→    }: {
   304→      id: string;
   305→      data: Record<string, unknown>;
   306→    }) => {
   307→      const res = await fetch(`/api/pinterest/accounts/${id}`, {
   308→        method: "PATCH",
   309→        headers: { "Content-Type": "application/json" },
   310→        body: JSON.stringify(data),
   311→      });
   312→      if (!res.ok) throw new Error("Failed to update");
   313→      return res.json();
   314→    },
   315→    onSuccess: () => {
   316→      qc.invalidateQueries({ queryKey: ["pinterest"] });
   317→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   318→    },
   319→    onError: () => toast({ title: "Update failed", variant: "destructive" }),
   320→  });
   321→
   322→  // Move account up/down in rotation queue
   323→  const moveAccount = useMutation({
   324→    mutationFn: async ({
   325→      account,
   326→      direction,
   327→    }: {
   328→      account: Account;
   329→      direction: "up" | "down";
   330→    }) => {
   331→      if (!accounts) return;
   332→      const sorted = [...accounts].sort((a, b) => a.orderIndex - b.orderIndex);
   333→      const idx = sorted.findIndex((a) => a.id === account.id);
   334→      const swapIdx = direction === "up" ? idx - 1 : idx + 1;
   335→      if (swapIdx < 0 || swapIdx >= sorted.length) return;
   336→      const swapWith = sorted[swapIdx];
   337→
   338→      // Swap orderIndex values
   339→      await fetch(`/api/pinterest/accounts/${account.id}`, {
   340→        method: "PATCH",
   341→        headers: { "Content-Type": "application/json" },
   342→        body: JSON.stringify({ orderIndex: swapWith.orderIndex }),
   343→      });
   344→      const res = await fetch(`/api/pinterest/accounts/${swapWith.id}`, {
   345→        method: "PATCH",
   346→        headers: { "Content-Type": "application/json" },
   347→        body: JSON.stringify({ orderIndex: account.orderIndex }),
   348→      });
   349→      if (!res.ok) throw new Error("Failed to move");
   350→    },
   351→    onSuccess: () => {
   352→      qc.invalidateQueries({ queryKey: ["pinterest"] });
   353→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   354→      toast({ title: "Rotation order updated" });
   355→    },
   356→    onError: () => toast({ title: "Failed to reorder", variant: "destructive" }),
   357→  });
   358→
   359→  // Delete account
   360→  const deleteAccount = useMutation({
   361→    mutationFn: async (id: string) => {
   362→      const res = await fetch(`/api/pinterest/accounts/${id}`, {
   363→        method: "DELETE",
   364→      });
   365→      if (!res.ok) throw new Error("Failed to delete");
   366→      return res.json();
   367→    },
   368→    onSuccess: () => {
   369→      qc.invalidateQueries({ queryKey: ["pinterest"] });
   370→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   371→      toast({ title: "Account removed" });
   372→    },
   373→    onError: () => toast({ title: "Delete failed", variant: "destructive" }),
   374→  });
   375→
   376→  const handleStatusChange = (acc: Account, status: string) => {
   377→    updateAccount.mutate({ id: acc.id, data: { status } });
   378→    toast({
   379→      title: `${acc.name} → ${status.toLowerCase()}`,
   380→      description:
   381→        status === "ACTIVE"
   382→          ? "Account resumed"
   383→          : status === "PAUSED"
   384→          ? "Account paused"
   385→          : "Account skipped for this cycle",
   386→    });
   387→  };
   388→
   389→  const handlePrioritize = (acc: Account) => {
   390→    updateAccount.mutate({ id: acc.id, data: { priority: acc.priority > 0 ? 0 : 1 } });
   391→    toast({
   392→      title: acc.priority > 0 ? "Priority removed" : "Account prioritized",
   393→    });
   394→  };
   395→
   396→  const handlePinsPerBatch = (acc: Account, value: number) => {
   397→    updateAccount.mutate({ id: acc.id, data: { pinsPerBatch: value } });
   398→    toast({ title: `Daily quantity set to ${value} pins` });
   399→  };
   400→
   401→  const accountOfDay = stats?.accountOfDay ?? null;
   402→
   403→  return (
   404→    <div className="flex flex-col gap-6">
   405→      <PageHeader
   406→        icon={<ImageIcon className="h-5 w-5" />}
   407→        title="Pinterest"
   408→        description="Manage 10–100+ Pinterest accounts with daily rotation. Each day one account is the focus — produce pins, publish, then advance to the next."
   409→      />
   410→
   411→      {/* KPI ROW */}
   412→      <div className="grid grid-cols-2 gap-3 md:grid-cols-3 lg:grid-cols-5">
   413→        {statsLoading ? (
   414→          Array.from({ length: 5 }).map((_, i) => <StatCardSkeleton key={i} />)
   415→        ) : (
   416→          <>
   417→            <StatCard
   418→              label="Account of the Day"
   419→              value={
   420→                <span className="truncate block max-w-full">
   421→                  {accountOfDay ? accountOfDay.name : "—"}
   422→                </span>
   423→              }
   424→              hint={
   425→                accountOfDay
   426→                  ? `${accountOfDay.pinsRemaining} of ${accountOfDay.pinsPerBatch} pins left`
   427→                  : "No active account"
   428→              }
   429→              icon={<Crown className="h-4 w-4" />}
   430→              accent="amber"
   431→            />
   432→            <StatCard
   433→              label="Pins Created"
   434→              value={formatNumber(stats?.kpis.pinsCreated ?? 0)}
   435→              hint={`${stats?.kpis.totalAccounts ?? 0} accounts`}
   436→              icon={<Pin className="h-4 w-4" />}
   437→              accent="violet"
   438→            />
   439→            <StatCard
   440→              label="Pins Published"
   441→              value={formatNumber(stats?.kpis.pinsPublished ?? 0)}
   442→              hint={`${stats?.kpis.pinsScheduled ?? 0} scheduled`}
   443→              icon={<TrendingUp className="h-4 w-4" />}
   444→              accent="emerald"
   445→            />
   446→            <StatCard
   447→              label="Avg CTR"
   448→              value={formatPercent(stats?.kpis.avgCtr ?? 0, 2)}
   449→              hint="clicks ÷ impressions"
   450→              icon={<MousePointerClick className="h-4 w-4" />}
   451→              accent="sky"
   452→            />
   453→            <StatCard
   454→              label="Total Outbound Clicks"
   455→              value={formatNumber(stats?.kpis.totalOutboundClicks ?? 0, {
   456→                compact: true,
   457→              })}
   458→              hint={`${formatNumber(stats?.kpis.totalImpressions ?? 0, {
   459→                compact: true,
   460→              })} impressions`}
   461→              icon={<Eye className="h-4 w-4" />}
   462→              accent="rose"
   463→            />
   464→          </>
   465→        )}
   466→      </div>
   467→
   468→      <Tabs defaultValue="accounts" className="gap-4">
   469→        <TabsList className="w-fit">
   470→          <TabsTrigger value="accounts" className="gap-1.5">
   471→            <Users className="h-3.5 w-3.5" /> Accounts
   472→          </TabsTrigger>
   473→          <TabsTrigger value="pins" className="gap-1.5">
   474→            <Pin className="h-3.5 w-3.5" /> Pins
   475→          </TabsTrigger>
   476→          <TabsTrigger value="performance" className="gap-1.5">
   477→            <Award className="h-3.5 w-3.5" /> Performance
   478→          </TabsTrigger>
   479→        </TabsList>
   480→
   481→        {/* ===================== ACCOUNTS TAB ===================== */}
   482→        <TabsContent value="accounts" className="flex flex-col gap-4">
   483→          {/* Account of the Day hero */}
   484→          <Card className="p-5 gap-4 bg-gradient-to-br from-amber-500/5 via-transparent to-transparent border-amber-500/30">
   485→            <div className="flex items-center gap-2">
   486→              <Crown className="h-4 w-4 text-amber-500" />
   487→              <span className="text-xs font-bold uppercase tracking-wider text-muted-foreground">
   488→                Account of the Day
   489→              </span>
   490→            </div>
   491→            {statsLoading ? (
   492→              <Skeleton className="h-32" />
   493→            ) : accountOfDay ? (
   494→              <div className="flex flex-col gap-4 lg:flex-row lg:items-stretch lg:justify-between">
   495→                <div className="flex-1 min-w-0">
   496→                  <div className="flex flex-wrap items-center gap-2">
   497→                    <h2 className="text-2xl font-bold">{accountOfDay.name}</h2>
   498→                    <StatusBadge status={accountOfDay.status} />
   499→                    {accountOfDay.priority > 0 && (
   500→                      <Badge variant="outline" className="border-amber-300 bg-amber-100 text-amber-700 dark:bg-amber-500/15 dark:text-amber-400">
   501→                        <Star className="h-3 w-3 mr-1 fill-current" /> Priority
   502→                      </Badge>
   503→                    )}
   504→                  </div>
   505→                  <div className="mt-1 flex flex-wrap items-center gap-x-4 gap-y-1 text-sm text-muted-foreground">
   506→                    <span>Niche: <span className="text-foreground font-medium">{accountOfDay.nicheName ?? "—"}</span></span>
   507→                    <span>Website: <span className="text-foreground font-medium">{accountOfDay.websiteName ?? "—"}</span></span>
   508→                    <span>Cycle: <span className="text-foreground font-medium">{accountOfDay.currentCycle}</span></span>
   509→                  </div>
   510→                  {accountOfDay.targetAudience && (
   511→                    <p className="mt-1 text-sm text-muted-foreground">
   512→                      Audience: <span className="text-foreground">{accountOfDay.targetAudience}</span>
   513→                    </p>
   514→                  )}
   515→
   516→                  {/* Progress */}
   517→                  <div className="mt-3 max-w-md">
   518→                    <div className="flex items-center justify-between text-xs mb-1">
   519→                      <span className="text-muted-foreground">
   520→                        {accountOfDay.pinsCompleted} / {accountOfDay.pinsPerBatch} pins completed
   521→                      </span>
   522→                      <span className="font-medium">{accountOfDay.pinsRemaining} left</span>
   523→                    </div>
   524→                    <Progress
   525→                      value={
   526→                        accountOfDay.pinsPerBatch > 0
   527→                          ? (accountOfDay.pinsCompleted / accountOfDay.pinsPerBatch) * 100
   528→                          : 0
   529→                      }
   530→                      className="h-2"
   531→                    />
   532→                  </div>
   533→
   534→                  {/* Mini stats grid */}
   535→                  <div className="mt-4 grid grid-cols-2 gap-2 sm:grid-cols-4 max-w-2xl">
   536→                    <MiniStat label="Pins Published" value={formatNumber(accountOfDay.pinsPublished)} />
   537→                    <MiniStat label="Last Worked" value={accountOfDay.lastWorkedDate ? formatRelative(accountOfDay.lastWorkedDate) : "—"} />
   538→                    <MiniStat label="Next Work" value={accountOfDay.nextWorkDate ? formatRelative(accountOfDay.nextWorkDate) : "—"} />
   539→                    <MiniStat label="Traffic" value={formatNumber(accountOfDay.trafficGenerated, { compact: true })} />
   540→                  </div>
   541→                </div>
   542→
   543→                <div className="flex flex-col gap-2 lg:w-56 lg:border-l lg:pl-5">
   544→                  <div className="rounded-lg border bg-card p-3">
   545→                    <div className="text-xs text-muted-foreground">Revenue generated</div>
   546→                    <div className="text-xl font-bold text-emerald-600 dark:text-emerald-400">
   547→                      {formatCurrency(accountOfDay.revenueGenerated)}
   548→                    </div>
   549→                  </div>
   550→                  <Button
   551→                    onClick={() => advance.mutate(accountOfDay.id)}
   552→                    disabled={advance.isPending}
   553→                    className="w-full"
   554→                  >
   555→                    <RefreshCw className={cn("h-4 w-4", advance.isPending && "animate-spin")} />
   556→                    Mark as Worked &amp; Advance
   557→                  </Button>
   558→                  <p className="text-[11px] text-muted-foreground text-center">
   559→                    Resets pinsCompleted, moves account to back of queue
   560→                  </p>
   561→                </div>
   562→              </div>
   563→            ) : (
   564→              <EmptyState
   565→                icon={<Crown className="h-8 w-8" />}
   566→                title="No active account"
   567→                description="Add or resume a Pinterest account to set today's focus."
   568→              />
   569→            )}
   570→          </Card>
   571→
   572→          {/* Rotation queue */}
   573→          <Card className="p-4 gap-3">
   574→            <SectionHeader
   575→              title="Rotation Queue"
   576→              description={`${accounts?.length ?? 0} account(s) — ordered by daily rotation. Drag-free up/down controls.`}
   577→              icon={<RefreshCw className="h-4 w-4" />}
   578→              actions={<AddAccountDialog niches={stats?.niches ?? []} websites={stats?.websites ?? []} />}
   579→            />
   580→            {accountsLoading ? (
   581→              <div className="flex flex-col gap-2">
   582→                {Array.from({ length: 5 }).map((_, i) => (
   583→                  <Skeleton key={i} className="h-12" />
   584→                ))}
   585→              </div>
   586→            ) : !accounts || accounts.length === 0 ? (
   587→              <EmptyState
   588→                icon={<Users className="h-8 w-8" />}
   589→                title="No Pinterest accounts yet"
   590→                description="Add your first account to start the daily rotation."
   591→                action={<AddAccountDialog niches={stats?.niches ?? []} websites={stats?.websites ?? []} />}
